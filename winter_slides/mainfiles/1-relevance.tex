\subsection{Расходы по памяти}

\begin{frame}
  \frametitle{Память для обучения}
  \begin{itemize}
    \item Современные модели содержат $N \gtrsim 10^9$ параметров.
    \item Ограничения памяти GPU:
      \begin{itemize}
        \item Веса: 
        \[
          M_w = N \cdot b, \quad \text{где } b \text{ --- байт на параметр.}
        \]
        \item Градиенты: 
        \[
          M_g = N \cdot b.
        \]
        \item Оптимайзер: 
        \[
          M_{\text{opt}} = c \cdot N \cdot b, \quad \text{где } c\approx2\text{–}3.
        \]
      \end{itemize}
  \end{itemize}
\end{frame}
\begin{frame}
  \frametitle{Память для обучения}
  \begin{itemize}
      \item Промежуточные активации: так же зависит от размера батча, количества слоев и скрытого состояния.
      \item Общая формула для обучения:
      \[
        M_{\text{train}} = N \cdot b \cdot (2 + c) + M_{\text{act}}.
      \]
  \end{itemize}
\end{frame}
